# Root scope: character being checked as a potential protector.
bwf_can_guarantee_border_protection_trigger = {
	save_temporary_scope_as = bwf_protection_candidate
	is_landed = yes
	highest_held_title_tier >= tier_duchy
	highest_held_title_tier < tier_hegemony
	sub_realm_size < border_war_sized_realm_value
	house ?= {
		NOT = { has_house_unity_stage = antagonistic }
	}
	NOT = {
		any_character_war = {
			primary_defender = scope:bwf_protection_candidate
			using_civil_war_cb_trigger = yes
		}
	}
}

# Root scope: potential protector.
# True when root qualifies for protection and no liege above root does,
# i.e. root is the highest eligible protector in the chain.
bwf_is_topmost_protection_candidate_trigger = {
	bwf_can_guarantee_border_protection_trigger = yes
	NOT = {
		any_liege_or_above = {
			bwf_can_guarantee_border_protection_trigger = yes
		}
	}
}

# Root scope: any.
# Shared precondition for the declaration-screen triggers.
# Required scopes: attacker, defender, target.
bwf_border_war_declaration_scopes_valid_trigger = {
	exists = scope:attacker
	exists = scope:defender
	exists = scope:target
	scope:defender = {
		is_landed = yes
	}
}

# Root scope: the top liege being checked.
# Required scopes: attacker, defender.
bwf_top_liege_allows_direct_vassal_attack_trigger = {
	this != scope:attacker.top_liege
	this != scope:defender
	can_vassals_be_attacked = yes
}

# Root scope: overlord being evaluated by the CB AI defensive-power hook.
# Required scopes:
# - scope:attacker, the character evaluating the war declaration
# - scope:defender, the selected defender/subject
bwf_border_war_ai_overlord_is_guaranteed_protector_trigger = {
	save_temporary_scope_as = bwf_ai_overlord_candidate
	exists = scope:attacker
	exists = scope:defender
	this != scope:defender
	bwf_can_guarantee_border_protection_trigger = yes
	scope:defender = {
		is_landed = yes
		is_vassal_or_below_of = scope:bwf_ai_overlord_candidate
	}
	scope:defender.top_liege = {
		bwf_top_liege_allows_direct_vassal_attack_trigger = yes
	}
}

# Declaration-screen scopes (root / scope:attacker, scope:defender, scope:target):
# Core search shared by the join and warning triggers.
# True when a highest eligible intermediate protector exists for the target,
# saving them as scope:bwf_expected_border_war_protector.
bwf_expected_border_war_protector_exists_trigger = {
	trigger_if = {
		limit = {
			bwf_border_war_declaration_scopes_valid_trigger = yes
		}
		scope:defender.top_liege = {
			bwf_top_liege_allows_direct_vassal_attack_trigger = yes
		}
		scope:target = {
			holder ?= {
				OR = {
					this = scope:defender
					is_vassal_or_below_of = scope:defender
				}
				any_liege_or_above = {
					this != scope:defender
					bwf_is_topmost_protection_candidate_trigger = yes
					save_temporary_scope_as = bwf_expected_border_war_protector
				}
			}
		}
	}
	trigger_else = {
		always = no
	}
}

# Declaration-screen scopes. True when a local protector will be added on war start.
bwf_border_war_higher_protector_will_join_trigger = {
	bwf_expected_border_war_protector_exists_trigger = yes
}

# Declaration-screen scopes. Emits the warning naming the exact local protector.
# Consumed inverted by the scripted GUI's is_valid so BuildTooltip prints the
# custom_description below; also reserved for future CB display-trigger wiring.
# Relies on the temporary scope saved by the core trigger remaining visible to
# the custom_description; verify the rendered tooltip name in-game.
bwf_border_war_higher_protector_warning_trigger = {
	bwf_expected_border_war_protector_exists_trigger = yes
	# Guarded so the subject scope is only dereferenced when actually saved.
	trigger_if = {
		limit = {
			exists = scope:bwf_expected_border_war_protector
		}
		custom_description = {
			text = bwf_border_war_protector_will_join_tt
			subject = scope:bwf_expected_border_war_protector
			always = yes
		}
	}
}
